cmake_minimum_required(VERSION 3.16)
project(tracer_navigation_driver)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tracer_msgs REQUIRED)
find_package(ugv_sdk REQUIRED)
add_executable(tracer_safe_base src/tracer_safe_base.cpp)
target_link_libraries(tracer_safe_base ugv_sdk)
ament_target_dependencies(tracer_safe_base rclcpp std_msgs geometry_msgs nav_msgs tracer_msgs)
install(TARGETS tracer_safe_base DESTINATION lib/${PROJECT_NAME})
ament_package()
